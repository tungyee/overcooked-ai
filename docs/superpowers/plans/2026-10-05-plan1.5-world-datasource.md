```python
"""可走性判定与实体去重：全部纯函数。

从 oc/ai/world.py 的 World.__init__ 抽取。原代码把 nodeMap 解析、边界
计算、AABB 占用、传送带格、三类去重全挤在一个构造函数里，单测必须造
完整快照。这里改为入参是 dict 与基本类型的纯函数，与 Plan 1 的
predicates / navigation 风格一致。

与旧实现的两处行为差异（均为修 bug，有测试钉住）：
  1. parse_node_map 校验位串长度，不符返回 None；旧代码直接下标会抛
     IndexError 把主循环带崩。
  2. playable_bounds 统一用 .get() is not None 判站点 gx/gz；旧代码在
     边界计算处用 "gx" in s（判键存在）、在占用计算处用 .get()（判非
     None），两种写法不一致。实测 71 站点里 10 个无 gx 键。
"""
from typing import Callable, Dict, List, Optional, Set, Tuple

# nodeMap 下标偏移不在这里定义：GridMapping 已持有 off_x / off_z
# （geometry.OFF_X_DEFAULT 是它的兜底）。同一个常量定义两处必然漂移，
# 本模块一律从调用方传入。


def parse_node_map(nm):
    # type: (Optional[str]) -> Optional[Tuple[List[List[bool]], int, int]]
    """解析 "ROWSxCOLS=位串" 为 (矩阵, rows, cols)。

    返回 None 表示"无可用 nodeMap"——调用方应退化为只看站点占用，
    而不是把 None 当"全不可走"。
    """
    if not nm or "=" not in nm:
        return None
    dims, bits = nm.split("=", 1)
    if "x" not in dims:
        return None
    try:
        rows_s, cols_s = dims.split("x", 1)
        rows, cols = int(rows_s), int(cols_s)
    except ValueError:
        return None
    if rows <= 0 or cols <= 0:
        return None
    # 长度必须精确匹配：短了会下标越界，长了说明维度声明与数据不一致，
    # 两种情况都不该猜。
    if len(bits) != rows * cols:
        return None
    matrix = [[bits[r * cols + c] == "1" for c in range(cols)]
              for r in range(rows)]
    return (matrix, rows, cols)
```

```python
def playable_bounds(stations):
    # type: (Optional[List[Dict]]) -> Optional[Tuple[int, int, int, int]]
    """用站点 (gx, gz) 的范围界定厨房可玩区，返回 (min_gx, max_gx, min_gz, max_gz)。

    为什么需要它：nodeMap 外围有一圈误标为 '1' 的格子（厨房围墙之外）。
    只看 nodeMap 会把厨师往墙外引，所以要用站点范围夹一道。

    只统计 gx 与 gz **都**有值的站点——单轴站点无法定位。
    """
    gxs = []  # type: List[int]
    gzs = []  # type: List[int]
    for st in stations or []:
        if not isinstance(st, dict):
            continue
        gx, gz = st.get("gx"), st.get("gz")
        if gx is None or gz is None:
            continue
        gxs.append(gx)
        gzs.append(gz)
    if not gxs:
        return None
    return (min(gxs), max(gxs), min(gzs), max(gzs))
```

```python
def station_blocked_cells(stations, mapping):
    # type: (Optional[List[Dict]], object) -> Set[Tuple[int, int]]
    """站点物理占用的不可走格集合。

    两条来源，取并集：
      1. Collider AABB（minx/maxx/minz/maxz）——能正确覆盖多格站点，
         如出餐口横跨两格。
      2. gx/gz 单格兜底——AABB 字段缺失时（旧录制格式）仍有结果。

    为什么不能只靠 nodeMap：ServerPlateStation 这类功能站不进游戏的
    占用表，nodeMap 把它标成可走，厨师会撞上去。
    """
    blocked = set()  # type: Set[Tuple[int, int]]
    half_x = mapping.cell / 2.0
    half_z = mapping.cell_z / 2.0
    for st in stations or []:
        if not isinstance(st, dict):
            continue
        gx0, gz0 = st.get("gx"), st.get("gz")
        if gx0 is not None and gz0 is not None:
            blocked.add((int(gx0), int(gz0)))
        minx, minz = st.get("minx"), st.get("minz")
        maxx, maxz = st.get("maxx"), st.get("maxz")
        if minx is None or minz is None or maxx is None or maxz is None:
            continue
        # 扫描范围取 AABB 所跨格区间再各外扩 1 格，确保边缘格也被检查
        gx_lo = int(math.floor((minx - mapping.origin_x) / mapping.cell)) - 1
        gx_hi = int(math.floor((maxx - mapping.origin_x) / mapping.cell)) + 1
        gz_lo = int(math.floor((minz - mapping.origin_z) / mapping.cell_z)) - 1
        gz_hi = int(math.floor((maxz - mapping.origin_z) / mapping.cell_z)) + 1
        for gxc in range(gx_lo, gx_hi + 1):
            cx, _ = mapping.to_world(gxc, 0)
            if not (cx + half_x > minx + _OVERLAP_EPS
                    and cx - half_x < maxx - _OVERLAP_EPS):
                continue
            for gzc in range(gz_lo, gz_hi + 1):
                _, cz = mapping.to_world(0, gzc)
                if (cz + half_z > minz + _OVERLAP_EPS
                        and cz - half_z < maxz - _OVERLAP_EPS):
                    blocked.add((gxc, gzc))
    return blocked
```

```python
def make_walkable(node_map, rows, cols, bounds, blocked, belt_cells,
                  off_x, off_z):
    # type: (Optional[List[List[bool]]], int, int, Optional[Tuple[int,int,int,int]], Set[Tuple[int,int]], Set[Tuple[int,int]], int, int) -> Callable[[int, int], bool]
    """返回 is_walkable(gx, gz) 闭包，供 navigation.distance_field 使用。

    返回闭包而非每次传八个参数：distance_field 每拍调用上千次，参数
    绑定一次就好。闭包捕获的全是不可变快照数据，无副作用。

    判定优先级（顺序有意义）：
      1. 站点物理占用 → 不可走。功能站 nodeMap 标可走但撞不过去。
      2. nodeMap 缺失 → 可走。宁可走错也不要让 AI 整体瘫掉。
      3. 越界（可玩边界外）→ 不可走。nodeMap 外围有一圈误标 '1' 的墙外格。
      4. 传送带 → 可走。nodeMap 标 '0' 但角色能走上去。
      5. 查 nodeMap，下标越界按不可走。
    """
    def is_walkable(gx, gz):
        # type: (int, int) -> bool
        if (gx, gz) in blocked:
            return False
        if node_map is None:
            return True
        if bounds is not None:
            min_gx, max_gx, min_gz, max_gz = bounds
            if not (min_gx <= gx <= max_gx and min_gz <= gz <= max_gz):
                return False
        if (gx, gz) in belt_cells:
            return True
        r, c = gx + off_x, gz + off_z
        if 0 <= r < rows and 0 <= c < cols:
            return node_map[r][c]
        return False

    return is_walkable
```

```python
# 位置去重的小数位数。插件两端上报同一对象时末位可能有浮点噪声。
_POS_ROUND = 2

# 判断烹饪台条目"信息量"的四个状态键
_COOKER_STATE_KEYS = ("isCooking", "cooked", "burning", "onFire")


def dedupe_cookers(raw):
    # type: (Optional[List]) -> List[Dict]
    """按位置去重烹饪台，同位置保留状态字段更完整的那条。

    插件会同时报 Client 与 Server 两个烹饪台对象。Client 那份常常只有
    位置，Server 那份才带 isCooking/cooked/burning/onFire——保留错了
    就永远读不到"锅好没好"。
    """
    def score(entry):
        # type: (Dict) -> int
        return sum(1 for k in _COOKER_STATE_KEYS if entry.get(k) is not None)

    by_pos = {}  # type: Dict[Tuple[float, float], Dict]
    for entry in raw or []:
        if not isinstance(entry, dict):
            continue
        x, z = entry.get("x"), entry.get("z")
        if x is None or z is None:
            continue
        key = (round(x, _POS_ROUND), round(z, _POS_ROUND))
        cur = by_pos.get(key)
        if cur is None or score(entry) > score(cur):
            by_pos[key] = entry
    return list(by_pos.values())
```

```python
def dedupe_items(raw):
    # type: (Optional[List]) -> List[Dict]
    """按 名字+位置 去重散落物品，保留首次出现顺序。

    键里带名字（灶不带）：两个不同食材可以叠在同一格，只按位置去重
    会吞掉一个。
    """
    seen = set()  # type: Set[Tuple[str, float, float]]
    out = []      # type: List[Dict]
    for entry in raw or []:
        if not isinstance(entry, dict):
            continue
        x, z = entry.get("x"), entry.get("z")
        if x is None or z is None:
            continue
        key = (str(entry.get("n") or ""),
               round(x, _POS_ROUND), round(z, _POS_ROUND))
        if key in seen:
            continue
        seen.add(key)
        out.append(entry)
    return out


def dedupe_extinguishers(raw):
    # type: (Optional[List]) -> List[Dict]
    """按位置去重灭火器，同位置优选名字不含 "rigidbody" 的那条。

    插件会报出同一个灭火器对象上的多个同名组件。Rigidbody 组件的位置
    是物理中心而非交互点，取错了会站错位拿不到。只有 Rigidbody 版本
    时仍保留——有总比没有好。
    """
    by_pos = {}  # type: Dict[Tuple[float, float], Dict]
    for entry in raw or []:
        if not isinstance(entry, dict):
            continue
        x, z = entry.get("x"), entry.get("z")
        if x is None or z is None:
            continue
        key = (round(x, _POS_ROUND), round(z, _POS_ROUND))
        cur = by_pos.get(key)
        cur_is_rigid = (cur is not None
                        and "rigidbody" in str(cur.get("n") or "").lower())
        new_is_rigid = "rigidbody" in str(entry.get("n") or "").lower()
        if cur is None or (cur_is_rigid and not new_is_rigid):
            by_pos[key] = entry
    return list(by_pos.values())
```

```python
_FRAME_SEP = b"\n"
_RECV_SIZE = 65536


def split_frames(buf):
    # type: (bytes) -> Tuple[List[bytes], bytes]
    """把字节缓冲按 \\n 切成完整帧，返回 (完整行列表, 剩余缓冲)。

    一次 recv 既可能带回半帧，也可能带回多帧——两种都要正确处理：
    半帧留在剩余缓冲里等补齐，多帧全部切出（漏切会让缓冲无限增长）。
    空行丢弃，免得下游 json.loads 抛。
    """
    if _FRAME_SEP not in buf:
        return ([], buf)
    *complete, rest = buf.split(_FRAME_SEP)
    return ([line for line in complete if line.strip()], rest)
```

```python
class StateClient(object):
    """按行读取插件推送的 JSON 帧。

    connect 参数用于注入假 socket：分帧的边界条件（半帧、多帧、EOF）
    在真 socket 上无法稳定复现——内核什么时候切包不由测试决定。
    """

    def __init__(self, host="127.0.0.1", port=27411, connect=None):
        # type: (str, int, Optional[Callable[[], object]]) -> None
        if connect is None:
            self.s = socket.create_connection((host, port), timeout=10)
        else:
            self.s = connect()
        self.buf = b""
        self._pending = []  # type: List[bytes]

    def read(self):
        # type: () -> Dict
        """返回下一个可用帧。坏帧跳过，socket 关闭时抛 EOFError。

        跳过而非抛出：录制与回放都见过截断行，运行时同样可能遇到。
        一行坏 JSON 不该终结整个会话。
        """
        while True:
            while self._pending:
                line = self._pending.pop(0)
                try:
                    obj = json.loads(line)
                except ValueError:
                    continue
                # 下游一律按 dict 访问 frame["..."]，标量/数组必须拦掉
                if isinstance(obj, dict):
                    return obj
            data = self.s.recv(_RECV_SIZE)
            if not data:
                raise EOFError("socket closed")
            self.buf += data
            self._pending, self.buf = split_frames(self.buf)

    def close(self):
        # type: () -> None
        self.s.close()
```

```python
def _dicts(raw):
    # type: (Optional[List]) -> List[Dict]
    """只保留列表里的 dict 条目。

    下游谓词层一律按 dict 访问 entry["..."]，标量混进来会在很远的
    地方炸，且堆栈看不出是数据问题。在入口拦掉。
    """
    return [e for e in (raw or []) if isinstance(e, dict)]


class World(object):
    """一帧快照的只读视图：解析 + 派生数据装配。

    构造函数只做三件事：解析快照列表、建 GridMapping、调 occupancy
    算派生数据并缓存。可走性与去重的实现都在 occupancy，寻路在
    navigation——本类不自己实现算法。

    与 oc/ai/world.py 的差异：
      - GRID_OX/GRID_OZ 类常量换成 GridMapping（从 grid.size 读格宽、
        用站点反解原点，换关卡免改代码）。
      - walkable 是 occupancy.make_walkable 返回的闭包，构造时绑定一次。
      - 寻路方法不在这里；调用方用 navigation.distance_field(w.walkable, ...)。
    """

    def __init__(self, state):
        # type: (Dict) -> None
        state = state or {}
        self.raw = state
        self.phase = state.get("phase")
        self.level = state.get("level")
        # 用 `is True`：插件漏发该字段时是 None，不能当成"已暂停"
        self.paused = state.get("paused") is True

        self.orders = _dicts(state.get("orders"))
        self.chefs = _dicts(state.get("chefs"))
        self.stations = _dicts(state.get("stations"))
        self.recipes = _dicts(state.get("recipes"))
        self.plates = _dicts(state.get("plates"))
        self.workstations = _dicts(state.get("workstations"))

        # 这三类插件会报重复，构造时就去重（各自规则不同，见 occupancy）
        self.cookers = occupancy.dedupe_cookers(state.get("cookers"))
        self.items = occupancy.dedupe_items(state.get("items"))
        self.extinguishers = occupancy.dedupe_extinguishers(
            state.get("extinguishers"))

        self.mapping = GridMapping.from_grid(state.get("grid"), self.stations)

        parsed = occupancy.parse_node_map(
            (state.get("grid") or {}).get("nodeMap"))
        if parsed is None:
            self._node_map, self._nm_rows, self._nm_cols = None, 0, 0
        else:
            self._node_map, self._nm_rows, self._nm_cols = parsed

        self.bounds = occupancy.playable_bounds(self.stations)
        self.station_blocked = occupancy.station_blocked_cells(
            self.stations, self.mapping)
        self.belt_cells, self.belt_dir = occupancy.belt_cells_and_dirs(
            self.level)

        self.walkable = occupancy.make_walkable(
            self._node_map, self._nm_rows, self._nm_cols,
            self.bounds, self.station_blocked, self.belt_cells,
            self.mapping.off_x, self.mapping.off_z,
        )

    def to_grid(self, x, z):
        # type: (float, float) -> Tuple[int, int]
        return self.mapping.to_grid(x, z)

    def to_world(self, gx, gz):
        # type: (int, int) -> Tuple[float, float]
        return self.mapping.to_world(gx, gz)

    def chef(self, name):
        # type: (str) -> Optional[Dict]
        for c in self.chefs:
            if c.get("name") == name:
                return c
        return None
```

```python
x/z 同样不保证存在。

不含 find_plate_counter / PLATE_COUNTERS：spec §9 记它归 Plan 3，
替代品是 Planner 的候选生成 + 认领，不是一个 find 函数。
"""
from typing import Dict, List, Optional, Set, Tuple

# 出餐口的站点类型名
PLATE_STATION = "plate_station"

# 盘子堆位置（干净盘所在柜台）。当前关卡由用户实测标注。
# ⚠️ 仅在 stations[] 查不到任何盘堆时作兜底，泛化需插件输出（spec §10.1）。
PLATE_STACKS = {
    "Sushi_1_4_2P": [(-1, 0), (1, 0), (-1, -1), (1, -1)],
}


def _pos(st):
    # type: (Dict) -> Optional[Tuple[float, float]]
    """读站点世界坐标，缺任一轴返回 None。"""
    x, z = st.get("x"), st.get("z")
    if x is None or z is None:
        return None
    return (x, z)


def _dist2(st, x, z):
    # type: (Dict, float, float) -> Optional[float]
    """到 (x, z) 的平方距离。无需开方——只用于比较大小。"""
    pos = _pos(st)
    if pos is None:
        return None
    return (pos[0] - x) ** 2 + (pos[1] - z) ** 2


def nearest_station(stations, t, x, z):
    # type: (Optional[List[Dict]], Optional[str], float, float) -> Optional[Dict]
    """找离 (x, z) 最近的 t 类站点。t=None 表示不限类型。"""
    best, best_d = None, None
    for st in stations or []:
        if not isinstance(st, dict):
            continue
        if t is not None and st.get("t") != t:
            continue
        d = _dist2(st, x, z)
        if d is None:
            continue
        if best_d is None or d < best_d:
            best, best_d = st, d
    return best


def find_station(stations, t, near=None):
    # type: (Optional[List[Dict]], str, Optional[Tuple[float, float]]) -> Optional[Dict]
    """找某类型站点；near=(x, z) 时取最近，否则取第一个。"""
    if near is not None:
        return nearest_station(stations, t, near[0], near[1])
    for st in stations or []:
        if isinstance(st, dict) and st.get("t") == t:
            return st
    return None
```

```python
def find_crate(stations, ingredient):
    # type: (Optional[List[Dict]], Optional[str]) -> Optional[Dict]
    """找出产指定食材的箱子（t == "crate" 且 ing 匹配）。

    不分大小写：配方名与快照名的大小写不保证一致，匹配不上会让
    "去拿米"整条链静默失败——没有报错，只是永远不动。

    if not ingredient:
        return None
    want = str(ingredient).lower()
    for st in stations or []:
        if not isinstance(st, dict) or st.get("t") != "crate":
            continue
        if str(st.get("ing") or "").lower() == want:
            return st
    return None
```

```python
_POS_ROUND = 2


def _is_plate_source(st):
    # type: (Dict) -> bool
    """盘堆出生点，或晾干架（洗完的盘在这里可取）。"""
    t = st.get("t")
    if t == "plate_stack":
        return True
    return t == "plate_return" and "drying" in str(st.get("n") or "").lower()


def find_plate_stack(world, near=None):
    # type: (object, Optional[Tuple[float, float]]) -> Optional[Dict]
    """找可拿干净盘的位置，多个取离 near 最近。

    三级退化：
      1. stations[] 里的盘堆与晾干架，**过滤出当前真有干净盘的**。
         盘被端走后出生点仍在快照里但已空；盲按距离选会选中空位，
         厨师走过去拿不到盘、整条做菜链静默卡死。
      2. 一个都没有 → PLATE_STACKS 实测坐标兜底（关卡硬编码）。
      3. 仍没有 → 退到出餐口。都没有才返回 None。

    干净盘判据：plates 里 count == 0（count 是盘上食材数，0 = 空盘）。
    """
    seen = set()  # type: Set[Tuple[float, float]]
    stacks = []   # type: List[Dict]
    for st in getattr(world, "stations", None) or []:
        if not isinstance(st, dict) or not _is_plate_source(st):
            continue
        pos = _pos(st)
        if pos is None:
            continue
        key = (round(pos[0], _POS_ROUND), round(pos[1], _POS_ROUND))
        if key in seen:
            continue
        seen.add(key)
        stacks.append(st)

    clean_pos = set()
    for p in getattr(world, "plates", None) or []:
        if not isinstance(p, dict) or (p.get("count") or 0) != 0:
            continue
        pos = _pos(p)
        if pos is not None:
            clean_pos.add((round(pos[0], _POS_ROUND),
                           round(pos[1], _POS_ROUND)))
    if clean_pos:
        stocked = [s for s in stacks
                   if (round(s["x"], _POS_ROUND),
                       round(s["z"], _POS_ROUND)) in clean_pos]
        if stocked:
            stacks = stocked

    if not stacks:
        # 兜底：关卡实测坐标。⚠️ 泛化需插件输出（spec §10.1）。
        for gx, gz in PLATE_STACKS.get(getattr(world, "level", None)) or []:
            x, z = world.to_world(gx, gz)
            stacks.append({"t": "plate_stack", "n": "plate_stack",
                           "x": x, "z": z})
    if not stacks:
        return find_station(getattr(world, "stations", None),
                            PLATE_STATION, near)
    if near is not None:
        best, best_d = None, None
        for st in stacks:
            d = _dist2(st, near[0], near[1])
            if d is None:
                continue
            if best_d is None or d < best_d:
                best, best_d = st, d
        return best
    return stacks[0]
```

```python
def _load_legacy(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod
```

```python
def test_distance_field_reaches_stations_from_chef(level_frame):
    """从厨师位置 flood-fill 应能到达大部分站点的交互位。

    可走性走 World.walkable。这里曾内联一份只看"站点单格 + 矩形边界"
    的手搓 is_walkable（Plan 1 时还没有 World），它不看 nodeMap、能
    穿墙，比真实可走性宽松（实测距离场 70 格 vs 61 格）。内联版与
    生产代码各走一路，生产代码退化了它照旧绿，故删除。
    """
    w = W.World(level_frame)
    stations = w.stations
    if not any(s.get("gx") is not None for s in stations):
        pytest.skip("样本站点无网格坐标")

    # 真实帧里 chefs[].slot 全为 null（见 ledger 发现 A），chef_by_slot 取不到；
    # 距离场只需一个有世界坐标的起点，取任一有坐标的厨师即可。
    chef = None
    for c in w.chefs:
        if c.get("x") is not None and c.get("z") is not None:
            chef = c
            break
    if chef is None:
        pytest.skip("样本里没有带世界坐标的厨师")
    start = w.to_grid(chef["x"], chef["z"])

    field = distance_field(start, w.walkable, w.belt_dir)

    assert field[start] == 0
    reachable = 0
    for st in stations:
        gx, gz = st.get("gx"), st.get("gz")
        if gx is None or gz is None:
            continue
        cells = interact_cells(gx, gz, w.walkable)
        if any(c in field for c in cells):
            reachable += 1

    assert reachable > 0, "没有任何站点可达，距离场或可走判定有问题"
```
